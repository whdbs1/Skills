import ctypes

libc = ctypes.CDLL(None)
libc.syscall.restype = ctypes.c_long
sc = libc.syscall

_cnt = [1]

def c_compile(code, opt="O2"):
    t = _cnt[0]; _cnt[0] += 1
    s = ("/dev/shm/src_%d.c" % t).encode()
    o = ("/dev/shm/lib_%d.so" % t).encode()

    def die(msg):
        sc(87, s); sc(87, o)
        tp_log(msg)
        exit()

    fd = sc(2, s, 577, 0o644)
    data = code.encode()
    sc(1, fd, data, len(data))
    sc(3, fd)

    ef = (ctypes.c_int * 2)()
    sc(22, ctypes.byref(ef))
    er, ew = ef
    sc(72, ew, 2, 1)

    pid = sc(57)

    if pid == 0:
        argv = (ctypes.c_char_p * 8)(b"/usr/bin/gcc", b"-shared", b"-fPIC",
                                      ("-%s" % opt).encode(), b"-o", o, s, None)
        sc(3, er)
        sc(59, b"/usr/bin/gcc", argv, None)
        sc(1, ew, b"1", 1); sc(3, ew); sc(60, 1)
        return

    if pid < 0:
        sc(3, er); sc(3, ew)
        die(pid)

    sc(3, ew)
    st = ctypes.c_int()
    sc(61, pid, ctypes.byref(st), 0, None)

    buf = ctypes.c_char()
    failed = sc(0, er, ctypes.byref(buf), 1) > 0 or (st.value >> 8) & 0xFF
    sc(3, er)
    if failed:
        die(st.value)

    try:
        lib = ctypes.CDLL(o.decode())
    except OSError:
        die("dlopen")

    sc(87, s); sc(87, o)
    return {"lib": lib, "funcs": {}}


def c_func(c, name, restype, argtypes):
    f = getattr(c["lib"], name)
    f.restype, f.argtypes = restype, argtypes
    c["funcs"][name] = f

def c_call(c, name, *args):
    return c["funcs"][name](*args)
